# Create bond interfaces for each leaf connection

ip link set bond1 type bond lacp_rate fast
ip link set eth3 down
ip link set eth3 master bond1
ip link set bond1 type bond ad_actor_system 00:00:00:00:00:01
ip link set eth3 up
ip link set bond1 up

#############################
# VRF Configuration - for tenant separation
#############################

# Create VRF for tenant routing

ip link add vrf1 type vrf table 1000
ip link set vrf1 up


#############################
# L3VNI Configuration - for Inter-VLAN Routing
#############################

# Create bridge for L3VNI

ip link add br1000 type bridge
ip link set br1000 master vrf1
ip addr add 10.255.255.1/24 dev br1000


ip link add vni1000 type vxlan id 1000 dstport 4789 local 10.0.0.2 nolearning
ip link set vni1000 master br1000
ip link set vni1000 type bridge_slave neigh_suppress on learning off
ip link set vni1000 up
ip link set br1000 up

# L2 VNI
